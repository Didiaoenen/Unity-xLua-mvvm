
local ClassImpl = require("OOP.ClassImpl")

class = setmetatable({}, {
    __call = function(pCls, ...)
        return pCls.New(...)
    end
})

class.New = function(pName, ...)
    
    local cls = ClassImpl.CreateClass(pName)
    
    ClassImpl.CreateClassTables(cls)

    ClassImpl.MakeInternalObjectMeta(cls)

    local clsMeta = {}
    setmetatable(cls, clsMeta)

    ClassImpl.CreateClassInherit(cls, {...})

    local is = ClassImpl.CreateClassIs(cls)
    local new = ClassImpl.CreateClassNew(cls)
    local delete = ClassImpl.CreateClassDelete(cls)
    ClassImpl.AttachClassFunctions(cls, is, new, delete)

    clsMeta.__index = ClassImpl.ClassGet
    clsMeta.__newindex = ClassImpl.ClassSet
    return cls
end
